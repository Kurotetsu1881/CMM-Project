// ===== BOEING 777 IPC TARAMA KODU v1 (Toolbox AIPC -> ipc_B777.json) · paralel, kaldığı yerden devam eder =====
// Kullanım: toolbox.boeing.com'da 777 AIPC belgesini (örn. D633W111-THY-xxxx) DocumentViewer'da açın (herhangi bir figür açık olsun).
// F12 > Console'a bu kodun tamamını yapıştırıp Enter'a basın. Sol altta durum bandı çıkar; bittiğinde "ipc_B777.json" otomatik iner.
// Kod yalnızca SİZİN oturumunuzda, SİZİN yetkinizle ve sadece okuma istekleri yapar (içindekiler, figür tablosu, parça kayıtları, uçak kaydı).
// Durdurmak için: window.__stop = true   (sonra aynı kodu tekrar çalıştırırsanız kaldığı yerden devam eder)
(async () => {
  // ===== AYARLAR =====
  const TYPE = 'B777';
  const ENGINE = 'GE90-100 SERIES';
  const FLEET_MSN = [40707, 40708, 40791, 40792, 40709, 40710, 40711, 40793, 40794, 40795, 40796, 40797, 44116, 44117, 44118, 60401, 44119, 44120, 44122, 44121, 44124, 44123, 44125, 44126, 44127, 44128, 44129, 44130, 60405, 60402, 41818, 41819, 42097, 60403, 60404, 65742, 65743, 65744, 66578, 66579, 66580, 35783, 32729, 32714, 70425, 70426, 70427, 70424];  // filo dosyasındaki 777 MSN'leri
  const CHAPTERS = 'ALL';     // örn. ['25','52'] ya da tümü için 'ALL'
  const MAX_CONC = 16;        // otomatik hızlanmanın üst sınırı (sunucu sorunsuzsa 2'şer artırır, hata görürse yarıya indirir)
  const CONCURRENCY = 6;      // başlangıçtaki eşzamanlı istek sayısı
  const SKIP_NOFLEET = true;  // true: filomuzdaki hiçbir uçağa uygulanmayan figürleri atlar
  const SAVE_ONLY_CMM = true; // true: dosyaya yalnızca CMM'i olan satırları yazar
  const RESUME = true;        // true: kesilen tarama kaldığı yerden devam eder (önbellek belge sürümüne bağlıdır)
  const ALLOW_PARTIAL = false; // false: eksik figür kalırsa dosya İNDİRİLMEZ; kodu tekrar çalıştırmanız yeterli
  const MAX_FIGS = 0;         // 0: sınırsız. DENEME için 30 yazın: yalnızca ilk 30 figürü tarar, dosya ipc_B777_DENEME.json iner (CMM'siz satırlar da yazılır)
  // ===================
  const DIAG = !!MAX_FIGS;
  const sleep = ms => new Promise(r => setTimeout(r, ms));
  const setAll = (arr, src) => { const t = Array.from(src); arr.length = 0; for (let i = 0; i < t.length; i++) arr.push(t[i]); };
  const clean = s => String(s || '').replace(/\s+/g, ' ').trim();
  const rangeStr = idx => {                                        // [0,1,2,5] -> "0-2,5"
    const a = [...new Set(idx)].sort((x, y) => x - y), out = [];
    for (let i = 0; i < a.length;) { let j = i; while (j + 1 < a.length && a[j + 1] === a[j] + 1) j++; out.push(j > i ? a[i] + '-' + a[j] : '' + a[i]); i = j + 1; }
    return out.join(',');
  };

  const O = location.origin;
  const PN = (location.href.match(/productNumber=([^;&?#]+)/) || [])[1];
  const ISS = (location.href.match(/issueDate=([^;&?#]+)/) || [])[1];
  if (!PN || !ISS) return console.error('Toolbox DocumentViewer sayfasında değilsiniz (adreste productNumber/issueDate yok). 777 AIPC belgesini açıp tekrar deneyin.');
  const band = document.createElement('div');
  band.style.cssText = 'position:fixed;left:8px;bottom:8px;z-index:2147483647;background:#c00;color:#fff;font:bold 14px sans-serif;padding:8px 12px;border-radius:6px;max-width:80vw';
  band.textContent = TYPE + ' IPC TARAMA BAŞLADI... (sekmeyi kapatmayın)'; document.body.appendChild(band);

  // ---- devam etme için küçük IndexedDB deposu ----
  const IDB = {
    open() { return this._d || (this._d = new Promise((res, rej) => { const r = indexedDB.open('ipcscan', 1);
      r.onupgradeneeded = () => r.result.createObjectStore('kv'); r.onsuccess = () => res(r.result); r.onerror = () => { this._d = null; rej(r.error); }; })); },
    async set(k, v) { try { const d = await this.open(); await new Promise(res => { const t = d.transaction('kv', 'readwrite'); t.objectStore('kv').put(v, k); t.oncomplete = res; t.onerror = res; }); } catch (e) {} },
    async all(prefix) { const out = new Map(); try { const d = await this.open();
      await new Promise(res => { const q = d.transaction('kv').objectStore('kv').openCursor(IDBKeyRange.bound(prefix, prefix + '￿'));
        q.onsuccess = () => { const c = q.result; if (c) { out.set(c.key, c.value); c.continue(); } else res(); }; q.onerror = () => res(); }); } catch (e) {} return out; },
    async clear(prefix) { try { const d = await this.open(); await new Promise(res => { const t = d.transaction('kv', 'readwrite'); const s = t.objectStore('kv');
      const q = s.openCursor(IDBKeyRange.bound(prefix, prefix + '￿')); q.onsuccess = () => { const c = q.result; if (c) { s.delete(c.key); c.continue(); } }; t.oncomplete = res; t.onerror = res; }); } catch (e) {} }
  };

  // ---- istek katmanı: 404 -> {notFound}; 429/5xx/ağ hatası -> bekleyip yeniden dener ----
  let nReq = 0, nErr = 0, pauseUntil = 0, lastSlow = 0, consecFail = 0, limit = CONCURRENCY, lastDown = 0, errMark = 0, lastStatus = 0;
  window.__stop = false;
  const call = async (url) => {
    for (let k = 0; k < 5; k++) {
      if (window.__stop) return null;
      const w = pauseUntil - Date.now(); if (w > 0) await sleep(w);
      try {
        nReq++;
        const h = {Accept: 'application/json, text/plain, */*'}; try { const t = sessionStorage.getItem('x-auth-token'); if (t) h['x-auth-token'] = t; } catch (e) {}
        const r = await fetch(url, {method: 'GET', credentials: 'include', headers: h});
        lastStatus = r.status;
        if (r.status === 404) { consecFail = 0; return {notFound: true}; }
        if (r.status === 429 || r.status >= 500) {
          const ra = +r.headers.get('retry-after') || 0;
          if (Date.now() - lastSlow > 3000) { lastSlow = Date.now(); lastDown = lastSlow; limit = Math.max(3, Math.floor(limit / 2)); pauseUntil = Math.max(pauseUntil, Date.now() + Math.max(ra * 1000, 1500)); }
          throw new Error(r.status);
        }
        if (!r.ok) throw new Error(r.status);
        const j = JSON.parse(await r.text());                      // oturum düşmüşse HTML döner -> burada hata verir
        consecFail = 0; return j;
      } catch (e) { nErr++; await sleep(400 * (k + 1)); }
    }
    if (++consecFail >= 25) { window.__stop = true; console.error('Art arda çok hata (son durum: ' + lastStatus + '): oturum süresi dolmuş olabilir. Sayfayı yenileyip (gerekirse tekrar giriş yapıp) kodu yeniden çalıştırın; kaldığı yerden devam eder.'); }
    return null;
  };
  let lastLog = 0, phase = '', total = 0, done = 0;
  const t0 = Date.now();
  const report = () => {
    if (Date.now() - lastLog < 5000) return; lastLog = Date.now();
    if (nErr === errMark && Date.now() - lastDown > 10000 && limit < MAX_CONC) limit = Math.min(MAX_CONC, limit + 2); errMark = nErr;
    band.textContent = `${TYPE} IPC · ${phase}: ${done}/${total} (durdurmak için konsola  window.__stop = true)`;
    console.log(`${phase}: ${done}/${total} · istek ${nReq} (${(nReq / ((Date.now() - t0) / 1000)).toFixed(1)}/sn) · hata ${nErr} · paralel ${limit}`);
  };
  const runPool = async (items, fn) => {
    let i = 0; done = 0; total = items.length;
    const worker = async id => {
      while (!window.__stop) {
        if (i >= items.length) return;
        if (id >= limit) { await sleep(250); continue; }
        const k = i++; if (k >= items.length) return;
        await fn(items[k], k); done++; report();
      }
    };
    await Promise.all(Array.from({length: MAX_CONC}, (_, id) => worker(id)));
  };
  const fail = msg => { band.textContent = '✖ ' + msg; console.error(msg); };

  // ---- 0) Belge (ürün örneği) ----
  const rev = await call(`${O}/product/productRevision/getPIRevisionByProductNbrAndIssueDateAndProductFormatLabelAndPIRevisionState?productNbr=${PN}&issueDate=${ISS}&productFormatLabel=XML&piRevisionState=PUBLISHED`);
  const pi = rev && rev.productInstanceId;
  if (!pi) return fail('Belge kaydı alınamadı (productInstanceId). Oturum açık mı? Sayfayı yenileyip tekrar deneyin.');

  const revDate = mm[1] ? `${mm[3]}-${MON[+mm[2] - 1]}-${mm[1]}` : '';
  const revNum = ((/-(\d+)$/.exec(PN) || [])[1] || '').replace(/^0+/, '') || String(rev.internalRevision || '');

  if (!RESUME) await IDB.clear('ipcB|');
  const cache = RESUME ? await IDB.all(KEY) : new Map();
  if (cache.size) console.log(`Önceki taramadan ${cache.size} kayıt hazır, tekrar indirilmeyecek.`);
  console.log(`%c${TYPE} IPC · ${PN} · ${revDate} · ürün: ${pi}`, 'font-weight:bold');

  // ---- 1) Filo: MSN -> Toolbox uçak kaydı (airplaneId, tescil, CEC) ----
  phase = 'filo';
  const raw = [], miss = [];
  await runPool(FLEET_MSN, async m => {
    const c = KEY + 'ap|' + m, hit = cache.get(c);
    if (hit) { raw.push(hit); return; }
    const j = await call(`${O}/metadata/findAirplaneBySerialNumber?serialNumber=${m}`);
    if (j && !j.notFound && j.id != null) {
      const a = {id: String(j.id), msn: String(j.serialNumber || m), reg: clean(j.currentRegistry || j.originalRegistry), cec: clean(j.customerEffectivityNumber || j.cec), st: clean(j.airplaneStatus), op: clean(j.currentOperatorCode)};
      raw.push(a); IDB.set(c, a);
    } else miss.push(m);
  });
  if (window.__stop) return fail('Durduruldu. Aynı kodu tekrar çalıştırırsanız kaldığı yerden devam eder.');
  if (miss.length) console.warn(`Toolbox'ta bulunamayan MSN'ler (${miss.length}): ${miss.join(', ')}`);
  const byReg = new Map();                                          // aynı tescil birden çok MSN'de görünürse aktif/THY olanı tut
  raw.forEach(a => {
    const o = byReg.get(a.reg), score = x => (/inactive/i.test(x.st) ? 0 : 2) + (x.op === 'THY' ? 1 : 0);
    if (!o) byReg.set(a.reg, a); else { console.warn(`Aynı tescil iki MSN'de: ${a.reg} (${o.msn} / ${a.msn})`); if (score(a) > score(o)) byReg.set(a.reg, a); }
  });
  const fleet = [...byReg.values()].filter(a => a.reg).sort((a, b) => (parseInt(a.cec, 10) || 1e9) - (parseInt(b.cec, 10) || 1e9) || a.msn.localeCompare(b.msn));
  if (!fleet.length) return fail('Filo çözülemedi (findAirplaneBySerialNumber sonuç vermedi).');
  const fidx = new Map(fleet.map((f, i) => [f.id, i]));
  console.log(`Filo: ${fleet.length} uçak`);

  // ---- 2) İçindekiler: bölümleri aç, figürleri topla ----
  const nodeOf = n => ({du: n.dataUnitId, key: n.locationKey || '', title: n.title || '', mt: n.moduleType || '', hc: !!n.hasChildren,
    ids: (n.applicItemInfo && n.applicItemInfo.applicableAirplaneIds) || null, txt: ((n.applicItemInfo && n.applicItemInfo.displayText) || []).join(',')});
  const flat = (arr, out = []) => { for (const n of arr || []) { out.push(n); if (n.childrenTOCNodeData && n.childrenTOCNodeData.length) flat(n.childrenTOCNodeData, out); } return out; };
  const tree = async du => {
    const ck = KEY + 'toc|' + (du || 'top'), hit = cache.get(ck); if (hit) return hit;
    const j = await call(`${O}/product/tocService/getTocTreeFromDuIdPiId?productInstanceId=${pi}` + (du ? `&dataUnitId=${du}` : ''));
    if (!Array.isArray(j)) return null;
    const ns = flat(j).filter(n => n.dataUnitId).map(nodeOf); IDB.set(ck, ns); return ns;
  };
  const chOk = k => CHAPTERS === 'ALL' || CHAPTERS.includes(k.slice(0, 2));
  const all = new Map(), expanded = new Set(), tocFail = [];
  const add = ns => ns.forEach(n => { if (!all.has(n.du)) all.set(n.du, n); });
  phase = 'içindekiler';
  let top = await tree('');
  if (!top || !top.length) {                                       // yedek: açık figürün kendi dataUnit'inden başla
    const lk = (location.href.match(/location=([0-9A-Za-z\-]+)/) || [])[1];
    const g = lk && await call(`${O}/product/dataUnitInstance/dataUnitsByProductInstanceIdAndLocationKey?productInstanceId=${pi}&locationKey=${lk}`);
    const du = g && g[0] && g[0].dataUnitId; top = du && await tree(du);
  }
  if (!top || !top.length) return fail('İçindekiler (getTocTreeFromDuIdPiId) alınamadı.');
  add(top);
  for (let guard = 0; guard < 8 && !window.__stop; guard++) {
    const queue = [...all.values()].filter(n => n.hc && n.mt !== 'figure' && /^\d\d(-\d\d)*$/.test(n.key) && !expanded.has(n.du) && chOk(n.key));
    if (!queue.length) break;
    phase = 'içindekiler (' + (guard + 1) + '. seviye)';
    await runPool(queue, async n => { expanded.add(n.du); const ns = await tree(n.du); if (ns) add(ns); else tocFail.push(n.key); });
  }
  if (window.__stop) return fail('Durduruldu. Aynı kodu tekrar çalıştırırsanız kaldığı yerden devam eder.');
  const sections = {}, chapters = [];
  all.forEach(n => {
    if (/^\d\d$/.test(n.key) && chOk(n.key)) chapters.push({ch: n.key, title: clean(n.title).replace(/^\d\d\s+/, '')});
    if (/^\d\d-\d\d$/.test(n.key) && chOk(n.key)) sections[n.key] = clean(n.title).replace(/^\d\d-\d\d\s+/, '');
  });
  chapters.sort((a, b) => a.ch.localeCompare(b.ch));
  let figs = [...all.values()].filter(n => n.mt === 'figure' && /^\d\d-/.test(n.key) && chOk(n.key));
  const nTree = figs.length;
  if (SKIP_NOFLEET) figs = figs.filter(n => !n.ids || n.ids.some(i => fidx.has(String(i))));
  figs.sort((a, b) => a.key.localeCompare(b.key));
  if (MAX_FIGS) { const step = Math.max(1, Math.floor(figs.length / MAX_FIGS)); figs = figs.filter((_, i) => i % step === 0).slice(0, MAX_FIGS); }
  console.log(`İçindekiler: ${nTree} figür, filomuza uygulanan ${figs.length}`);

  // ---- 3) Figürler: tablo + (CMM'li satır varsa) parça kayıtları ----
  const typeSeen = {};
  const walk = (o, f, d = 0) => { if (!o || typeof o !== 'object' || d > 12) return; f(o); (Array.isArray(o) ? o : Object.values(o)).forEach(v => walk(v, f, d + 1)); };
  const doFig = async n => {
    const ck = KEY + (DIAG ? 'fgD2|' : 'fg2|') + n.du, hit = cache.get(ck);
    if (hit) { res.set(n.du, hit); return; }
    const meta = await call(`${O}/productv2read/contentObject/primaryContentObjectByDataUnitAndProductInstanceId/${pi}/dataUnit/${n.du}/contentRefType/DISPLAYABLE`);
    if (!meta) { figFail.push(n.key); return; }
    let r = {key: n.key, rows: [], nRows: 0, nCmmRows: 0, unres: 0};
    const href = meta._links && meta._links.content && meta._links.content.href;
    if (meta.notFound || !href) r.why = 'icerik-yok';
    else {
      const tab = await call(href);
      if (!tab) { figFail.push(n.key); return; }
      if (!Array.isArray(tab)) r.why = 'tablo-formati';
      else {
        const parts = tab.map(x => {
          const fp = x.figurePartInfo || {}, pm = fp.pnrmfr || {}, noms = x.nomenclatures || [];
          const cm = noms.filter(z => z.name === 'ComponentMaintManualRef' || /COMPONENT MAINT/i.test(z.displayLabel || ''));
          const ein = noms.filter(z => z.name === 'einmfr').map(z => z.valueWithoutManufacturer || z.value).filter(Boolean);
          return {id: fp.itemId || fp.applicRefId || '', item: clean(fp.displayItemNumber || fp.itemNumber), pn: clean(pm.pnr), v: clean(pm.mfr), ref: String(x.pnrRef || '').toLowerCase(),
            qty: clean(x.quantityPerAssembly), adt: clean(x.adt), fin: ein.join(', '), ext: cm.flatMap(z => (z.externalCirRefs || []).map(e => e.externalPubIdent)).filter(Boolean), hasCmm: cm.length > 0};
        });
        r.nRows = parts.length; r.nCmmRows = parts.filter(p => p.hasCmm).length;
        if (parts.length) {
          // CMM referansı parça düzeyindedir: figür tablosunda değil, parça kayıtlarında (PART-CIR) bulunur -> her figür için parça kayıtları alınır
          const cir = await call(`${O}/productv2read/cirItem/cirItemByProductInstanceIdAndDataUnitId/productInstanceId/${pi}/dataUnitId/${n.du}?airplaneId=`);
          if (!cir) { figFail.push(n.key); return; }
          const ext = new Map(), kwd = new Map(), app = new Map(), pcirs = [];
          walk(cir, o => {
            if (o.externalRefId && o.externalPubReferenceItem) ext.set(o.externalRefId, o);
            if (o.cirModuleId === 'PART-CIR' && o.pnrmfr) { pcirs.push(o); if (o.kwd != null) kwd.set('k' + String(o.pnrmfr.mfr || '').toLowerCase() + String(o.pnrmfr.pnr || '').toLowerCase(), clean(o.kwd)); }
            if (Array.isArray(o.cirIdentifierAlias) && o.applicItemInfo && Array.isArray(o.applicItemInfo.applicableAirplaneIds))
              o.cirIdentifierAlias.forEach(a => { const s = app.get(a) || new Set(); o.applicItemInfo.applicableAirplaneIds.forEach(i => s.add(String(i))); app.set(a, s); });
          });
          const cmmOfRef = new Map();                              // 'k'+vendor+pn -> [{n,s}]
          const takeExt = (e, out) => {
            const it = e.externalReferenceCIRItem || ext.get(e.externalPubIdent), x = it && it.externalPubReferenceItem; if (!x) return;
            const ac = clean(x.sourceSystemDocAcronym); typeSeen[ac] = (typeSeen[ac] || 0) + 1;
            if (/CMM/i.test(ac) && x.userNameSpace) out.push({n: clean(x.userNameSpace), s: clean(x.supplierRefId || it.cageCode)});
          };
          pcirs.forEach(o => {
            const out = [];
            (o.nomenclatures || []).forEach(z => { if (z.name === 'ComponentMaintManualRef' || /COMPONENT MAINT/i.test(z.displayLabel || '')) (z.externalCirRefs || []).forEach(e => takeExt(e, out)); });
            if (out.length) cmmOfRef.set('k' + String(o.pnrmfr.mfr || '').toLowerCase() + String(o.pnrmfr.pnr || '').toLowerCase(), out);
          });
          parts.forEach(p => {
            const cm = (cmmOfRef.get('k' + p.v.toLowerCase() + p.pn.toLowerCase()) || []).slice();
            p.ext.forEach(e => takeExt({externalPubIdent: e}, cm));
            const uniq = []; cm.forEach(c => { if (!uniq.some(u => u.n === c.n && u.s === c.s)) uniq.push(c); });
            if (p.hasCmm && !uniq.length) r.unres++;
            if (!uniq.length && !DIAG) return;
            const k = kwd.get('k' + p.v.toLowerCase() + p.pn.toLowerCase()) || '';
            const ids = app.get(p.id);
            r.rows.push({item: p.item, pn: p.pn, v: p.v, nom: k ? (p.adt ? k + ' - ' + p.adt : k) : p.adt, fin: p.fin, qty: p.qty, ids: ids ? [...ids] : null, cmm: uniq, flag: uniq.length ? 'cmm-var' : (p.hasCmm ? 'cmm-cozulemedi' : 'cmm-yok')});
          });
        }
      }
    }
    res.set(n.du, r); IDB.set(ck, r);
  };
  const res = new Map(), figFail = [];
  phase = 'figürler';
  await runPool(figs, doFig);
  if (window.__stop) return fail('Durduruldu. Aynı kodu tekrar çalıştırırsanız kaldığı yerden devam eder.');
  const unres = [...res.values()].reduce((s, r) => s + (r.unres || 0), 0);
  if (!ALLOW_PARTIAL && (figFail.length || tocFail.length)) {
    band.textContent = 'Tarama eksik kaldı: kodu tekrar çalıştırın (yalnızca eksikler indirilir).';
    return console.error(`✖ Tarama eksik kaldı (alınamayan: ${figFail.length} figür, ${tocFail.length} ağaç düğümü). Dosya indirilmedi. Aynı kodu tekrar çalıştırın: yalnızca eksikler indirilir.`);
  }
  if (unres) console.warn(`${unres} satırda "COMPONENT MAINT MANUAL REF" var ama CMM numarası çözülemedi (dosyada yok).`);
  console.log('CMM kaynak türleri (sourceSystemDocAcronym):', typeSeen);

  // ---- 4) Çıktıyı derle ve indir ----
  const fmOf = ids => { const ix = ids.map(i => fidx.get(String(i))).filter(i => i != null); return ix.length >= fleet.length ? '*' : rangeStr(ix); };
  const outFigs = [], outRows = [];
  figs.forEach(n => {
    const r = res.get(n.du); if (!r) return;
    if (!r.rows.length && !DIAG) return;
    const fm = n.ids ? fmOf(n.ids) : '*';
    const fi = outFigs.length;
    outFigs.push({id: n.key, ref: n.key, name: clean(n.title).replace(n.key, '').trim(), zones: '', app: n.txt ? 'CEC ' + n.txt : '', fm, ch: n.key.slice(0, 2)});
    r.rows.forEach(w => {
      let m = '';
      if (w.ids) { const x = fmOf(w.ids); m = x === '' ? '-' : (x === fm ? '' : x); }
      const row = [fi, w.item, w.pn, w.v, w.nom, w.fin, w.qty, m, [...new Set(w.cmm.map(c => c.n))].join(','), ''];
      if (DIAG) row.push(w.flag);
      outRows.push(row);
    });
  });
  let rowsOut = outRows, figsOut = outFigs;
  if (SAVE_ONLY_CMM && !DIAG) { /* CMM'siz satırlar zaten eklenmedi */ }
  const cmmSup = {};
  res.forEach(r => r.rows.forEach(w => w.cmm.forEach(c => {
    const a = cmmSup[c.n] ? cmmSup[c.n].split(',') : []; if (c.s && !a.includes(c.s)) a.push(c.s); cmmSup[c.n] = a.join(',');
  })));
